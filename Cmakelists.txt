cmake_minimum_required(VERSION 3.16)
project(rapidserve VERSION 1.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

add_executable(rapidserve
    src/main.cpp
    src/server.cpp
    src/connection.cpp
    src/http_parser.cpp
    src/router.cpp
    src/file_cache.cpp
)

target_include_directories(rapidserve PRIVATE src)

find_package(Threads REQUIRED)
target_link_libraries(rapidserve PRIVATE Threads::Threads)

target_compile_options(rapidserve PRIVATE -O3 -march=native -flto)
target_link_options(rapidserve PRIVATE -flto)
